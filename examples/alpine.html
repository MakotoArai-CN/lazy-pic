    }
  };
}
</script>
Alpine.js - 响应式配置切换 
<div x-data="lazyPicDemo()" x-init="init()">
  <p>当前动画: <code x-text="placeholderAnimation"></code></p>
  <button @click="toggleAnimation()">
    切换 shimmer / progress-bar
  </button>

  <img class="lazy-image"
    data-src="photo.jpg" />
</div>

<script>
function lazyPicDemo() {
  return {
    instance: null,
    placeholderAnimation: 'shimmer',

    init() {
      this.mountLazyPic();
    },

    mountLazyPic() {
      // 先清理旧实例
      this.instance?.destroy();

      this.instance = new LazyPic.LazyPic({
        selector: '.lazy-image',
        strategy: 'animation',
        animationType: 'fade',
        animationDuration: 900,
        placeholder: {
          animation: this.placeholderAnimation,
          color: '#8b5cf6',
          showText: true,
          loadingText: 'Alpine 正在加载...'
        }
      });
      this.instance.init();
    },

    toggleAnimation() {
      this.placeholderAnimation =
        this.placeholderAnimation === 'shimmer'
          ? 'progress-bar'
          : 'shimmer';
      this.mountLazyPic();
    }
  };
}
</script>

Alpine.js - 资源清理 
// 方式一：beforeunload 清理
window.addEventListener('beforeunload', () => {
  const root = document.querySelector('[x-data]');
  const stack = root?._x_dataStack?.[0];
  stack?.instance?.destroy();
});

// 方式二：Alpine.plugin 统一管理
document.addEventListener('alpine:init', () => {
  Alpine.directive('lazypic', (el, { expression }, { cleanup }) => {
    const instance = new LazyPic.LazyPic(
      JSON.parse(expression)
    );
    instance.init();

    // Alpine 自动调用 cleanup
    cleanup(() => instance.destroy());
  });
});

// 使用自定义指令
// <img x-lazypic='{"selector":".lazy","strategy":"animation"}' />

API 方法 - 实例操作 
function lazyPicDemo() {
  return {
    instance: null,

    // 手动加载特定图片
    loadSingle(el) {
      this.instance?.loadImage(el);
    },

    // 更新配置（不销毁重建）
    updateAnim() {
      this.instance?.updateConfig({
        animationType: 'zoom',
        animationDuration: 800
      });
    },

    // 刷新（重新扫描 DOM 元素）
    rescan() {
      this.instance?.refresh();
    },

    // 获取当前加载状态
    checkStatus() {
      const s = this.instance?.getStatus();
      console.log('总数:', s.total, '已加载:', s.loaded);
    },

    // 销毁实例
    cleanup() {
      this.instance?.destroy();
      this.instance = null;
    }
  };
}
