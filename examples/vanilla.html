代码示例

基础使用 - ES Modules 
import { LazyPic } from 'lazy-pic';

const lazyPic = new LazyPic({
  selector: '.lazy-image',
  strategy: 'data-src',
  animationType: 'fade',
  animationDuration: 1000,
  enableBlur: true,
  blurIntensity: 8
});

lazyPic.init();
双图片策略配置 
const lazyPic = new LazyPic({
  selector: '.main-image',
  strategy: 'dual-image',
  animationType: 'slide',
  animationDuration: 1200,
  easing: 'ease-out',
  completionEffect: { enabled: false },
  rootMargin: '100px',
  threshold: 0.1,
  onStartLoad: (element) => {
    console.log('开始加载:', element.src);
  },
  onLoad: (element) => {
    console.log('加载完成:', element.src);
  },
  onError: (error, element) => {
    console.error('加载失败:', error.message);
  }
});

高级配置示例 
const lazyPic = new LazyPic({
  selector: '.animated-image',
  strategy: 'animation',
  animationType: 'zoom',
  animationDuration: 1500,
  easing: 'elastic',
  completionEffect: {
    enabled: true,
    type: 'glow',
    duration: 600
  },
  placeholder: {
    animation: 'shimmer',
    color: '#667eea',
    backgroundColor: 'rgba(248, 249, 250, 0.95)',
    showText: true,
    loadingText: '正在加载精美图片...',
    customContent: 'Loading preview'
  }
});
API 方法示例 
// 初始化
lazyPic.init();

// 手动加载特定图片
const imageElement = document.querySelector('.specific-image');
lazyPic.loadImage(imageElement);

// 更新配置
lazyPic.updateConfig({
  animationType: 'slide',
  animationDuration: 800
});

// 刷新并重新扫描元素
lazyPic.refresh();

// 获取当前状态
const status = lazyPic.getStatus();
console.log('加载状态:', status);

// 销毁实例
lazyPic.destroy();
