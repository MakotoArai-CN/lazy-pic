当策略、动画类型或占位符变化时，可通过 $: 语句自动清理旧实例并重新初始化，让配置与 DOM 始终同步。

在 onDestroy 中调用 destroy()，避免路由切换、条件渲染或列表卸载时残留 Observer 与占位层。

代码示例

Svelte 组件 - onMount 初始化 
<script>
  import { onMount, onDestroy } from 'svelte';
  import { LazyPic } from 'lazy-pic';

  let instance;

  onMount(() => {
    instance = new LazyPic({
      selector: '.svelte-lazy-image',
      strategy: 'animation',
      animationType: 'fade',
      animationDuration: 900,
      placeholder: {
        animation: 'shimmer',
        color: '#ff3e00',
        showText: true,
        loadingText: 'Svelte 正在加载图片...'
      }
    });

    instance.init();
  });

  onDestroy(() => {
    instance?.destroy();
  });
</script>
响应式声明 - $: 自动重建 
<script>
  import { onMount, onDestroy, tick } from 'svelte';
  import { LazyPic } from 'lazy-pic';

  let instance;
  let strategy = 'animation';
  let placeholderAnimation = 'shimmer';

  async function mountLazyPic() {
    instance?.destroy();
    await tick();

    instance = new LazyPic({
      selector: '.svelte-lazy-image',
      strategy,
      animationType: 'fade',
      placeholder: {
        animation: placeholderAnimation,
        color: '#ff3e00'
      }
    });

    instance.init();
  }

  onMount(mountLazyPic);

  $: if (instance) {
    strategy;
    placeholderAnimation;
    mountLazyPic();
  }

  onDestroy(() => instance?.destroy());
</script>

高级配置 - 动画策略 
const lazyPic = new LazyPic({
  selector: '.svelte-lazy-image',
  strategy: 'animation',
  animationType: 'zoom',
  animationDuration: 1400,
  easing: 'ease-out',
  completionEffect: {
    enabled: true,
    type: 'glow',
    duration: 500
  },
  placeholder: {
    animation: 'radar',
    color: '#ff3e00',
    backgroundColor: 'rgba(255,255,255,0.96)',
    showText: true,
    loadingText: 'Svelte 组件正在初始化图片...'
  },
  onLoad: (element) => {
    console.log('Svelte 图片加载完成', element.dataset.src || element.src);
  }
});
