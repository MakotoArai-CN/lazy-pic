Vue 3 Composition API - 基础用法 
import { LazyPic } from 'lazy-pic';
import { ref, onMounted, onBeforeUnmount } from 'vue';

export default {
  setup() {
    const instance = ref(null);

    onMounted(() => {
      instance.value = new LazyPic({
        selector: '.lazy-image',
        strategy: 'animation',
        animationType: 'fade',
        animationDuration: 900,
        placeholder: {
          animation: 'skeleton',
          color: '#42b883',
          showText: true,
          loadingText: '加载中...'
        }
      });
      instance.value.init();
    });

    onBeforeUnmount(() => {
      instance.value?.destroy();
    });
  }
};
Vue 3 - watch 响应式重建 
import { LazyPic } from 'lazy-pic';
import { ref, watch, onMounted, onBeforeUnmount } from 'vue';

export default {
  setup() {
    const instance = ref(null);
    const strategy = ref('animation');
    const animationType = ref('fade');

    const mountLazyPic = () => {
      instance.value?.destroy();
      instance.value = new LazyPic({
        selector: '.lazy-image',
        strategy: strategy.value,
        animationType: animationType.value,
        animationDuration: 1200,
        enableBlur: true,
        blurIntensity: 8,
        placeholder: {
          animation: 'shimmer',
          color: '#42b883'
        }
      });
      instance.value.init();
    };

    onMounted(mountLazyPic);

    watch([strategy, animationType], mountLazyPic);

    onBeforeUnmount(() => {
      instance.value?.destroy();
    });

    return { strategy, animationType };
  }
};
Vue 3 - 自定义组合函数 (Composable) 
// composables/useLazyPic.js
import { ref, onMounted, onBeforeUnmount, watch } from 'vue';
import { LazyPic } from 'lazy-pic';

export function useLazyPic(config) {
  const instance = ref(null);
  const status = ref(null);

  const mount = () => {
    instance.value?.destroy();
    instance.value = new LazyPic({
      ...config.value,
      onLoad: (el) => {
        status.value = instance.value.getStatus();
        config.value.onLoad?.(el);
      }
    });
    instance.value.init();
  };

  onMounted(mount);
  watch(config, mount, { deep: true });
  onBeforeUnmount(() => instance.value?.destroy());

  return {
    instance,
    status,
    refresh: () => instance.value?.refresh(),
    loadImage: (el) => instance.value?.loadImage(el)
  };
}

      v-for="img in images"
      :key="img.id"
      :data-src="img.fullSrc"
      class="lazy-image"
    />
  </div>

  <!-- 数据源策略 -->

      v-for="img in images"
      :key="img.id"
      :src="img.thumbSrc"
      :data-src="img.fullSrc"
      class="lazy-image"
    />
  </div>

  <!-- 双图片策略 -->
  <div v-for="img in images" :key="img.id"
    style="position: relative">
    <img :src="img.thumbSrc" class="thumbnail"
      style="position: absolute; width: 100%;" />
    <img :src="img.fullSrc" class="main-image"
      style="position: relative; z-index: 2;" />
  </div>
</template>
