React - useEffect 基础用法 
import { useEffect, useRef } from 'react';
import { LazyPic } from 'lazy-pic';

function Gallery() {
  const lazyPicRef = useRef(null);

  useEffect(() => {
    lazyPicRef.current = new LazyPic({
      selector: '.lazy-image',
      strategy: 'animation',
      animationType: 'fade',
      animationDuration: 900,
      placeholder: {
        animation: 'skeleton',
        color: '#0ea5e9',
        showText: true,
        loadingText: '加载中...'
      }
    });
    lazyPicRef.current.init();

    return () => {
      lazyPicRef.current?.destroy();
    };
  }, []);

  return (
    <div className="gallery">
      <img data-src="photo1.jpg" className="lazy-image" />
      <img data-src="photo2.jpg" className="lazy-image" />
    </div>
  );
}
React - 依赖驱动重建 
import { useEffect, useRef, useState } from 'react';
import { LazyPic } from 'lazy-pic';

function Gallery() {
  const lazyPicRef = useRef(null);
  const [animation, setAnimation] = useState('skeleton');
  const [strategy, setStrategy] = useState('animation');

  useEffect(() => {
    // cleanup 上一次的实例
    lazyPicRef.current?.destroy();

    lazyPicRef.current = new LazyPic({
      selector: '.lazy-image',
      strategy,
      animationType: 'fade',
      animationDuration: 1200,
      enableBlur: strategy === 'data-src',
      blurIntensity: 8,
      placeholder: {
        animation,
        color: '#0ea5e9'
      }
    });
    lazyPicRef.current.init();

    return () => {
      lazyPicRef.current?.destroy();
    };
  }, [animation, strategy]);

  return (
    <div>
      <button onClick={() =>
        setAnimation(a => a === 'skeleton' ? 'shimmer' : 'skeleton')
      }>切换动画</button>
      <div className="gallery">
        <img data-src="photo.jpg" className="lazy-image" />
      </div>
    </div>
  );
}
React - 自定义 Hook (useLazyPic) 
// hooks/useLazyPic.js
import { useEffect, useRef, useState } from 'react';
import { LazyPic } from 'lazy-pic';

export function useLazyPic(config) {
  const instanceRef = useRef(null);
  const [status, setStatus] = useState(null);

  useEffect(() => {
    instanceRef.current?.destroy();

    instanceRef.current = new LazyPic({
      ...config,
      onLoad: (el) => {
        setStatus(instanceRef.current.getStatus());
        config.onLoad?.(el);
      }
    });
    instanceRef.current.init();

    return () => instanceRef.current?.destroy();
  }, [config.strategy, config.animationType,
      config.placeholder?.animation]);

  return {
    instance: instanceRef,
    status,
    refresh: () => instanceRef.current?.refresh(),
    loadImage: (el) => instanceRef.current?.loadImage(el)
  };
}

// 使用
function App() {
  const { status, refresh } = useLazyPic({
    selector: '.lazy-image',
    strategy: 'animation',
    animationType: 'fade'
  });

  return <div>已加载: {status?.loaded}</div>;
}

CDN 方式 - 无 JSX 编译器 
<script src="https://unpkg.com/react@18/umd/react.production.min.js"></script>
<script src="https://unpkg.com/react-dom@18/umd/react-dom.production.min.js"></script>
<script src="https://unpkg.com/lazy-pic/dist/lazy-pic.umd.js"></script>
<script>
const { useEffect, useRef } = React;
const h = React.createElement;

function App() {
  const ref = useRef(null);

  useEffect(() => {
    ref.current = new LazyPic.LazyPic({
      selector: '.lazy-image',
      strategy: 'animation',
      animationType: 'fade',
      placeholder: { animation: 'skeleton', color: '#0ea5e9' }
    });
    ref.current.init();
    return () => ref.current?.destroy();
  }, []);

  return h('img', {
    'data-src': 'image.jpg',
    className: 'lazy-image'
  });
}

ReactDOM.createRoot(document.getElementById('root'))
  .render(h(App));
</script>
